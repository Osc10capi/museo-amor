socket.on('joinGame', (playerData) => { players[socket.id] = { id: socket.id, name: playerData.name, role: playerData.role, color: playerData.color || '#321827', x: playerData.role === 'oscar' ? 200 : 300, y: 0, direction: 1, isWalking: false, isCrouched: false, chatMessage: "", chatTimer: 0 }; io.emit('currentPlayers', players); }); socket.on('playerMovement', (movementData) => { if (players[socket.id]) { players[socket.id].x = movementData.x; players[socket.id].y = movementData.y; players[socket.id].direction = movementData.direction; players[socket.id].isWalking = movementData.isWalking; players[socket.id].isCrouched = movementData.isCrouched; players[socket.id].color = movementData.color; socket.broadcast.emit('playerMoved', players[socket.id]); } });
